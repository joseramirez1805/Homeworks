1. at(index)
const arr = [10, 20, 30];
console.log(arr.at(1));    // 20
console.log(arr.at(-1));   // 30 (último elemento)

2. slice(start, end)
const arr = [1, 2, 3, 4, 5];
console.log(arr.slice(1, 4)); // [2, 3, 4]

3. forEach(callback)
const arr = ['a', 'b', 'c'];
arr.forEach((el, i) => console.log(`Índice ${i}: ${el}`));

4. map(callback)
const nums = [1, 2, 3];
const dobles = nums.map(n => n * 2);
console.log(dobles); // [2, 4, 6]

5. entries()
const arr = ['x', 'y'];
for (const [i, val] of arr.entries()) {
  console.log(i, val); // 0 'x', 1 'y'
}

6. keys()
const arr = ['a', 'b'];
for (const key of arr.keys()) {
  console.log(key); // 0, 1
}

7. values()
const arr = ['a', 'b'];
for (const val of arr.values()) {
  console.log(val); // 'a', 'b'
}

8. find(callback)
const arr = [5, 12, 8];
const found = arr.find(x => x > 10);
console.log(found); // 12

9. findIndex(callback)
const arr = [5, 12, 8];
console.log(arr.findIndex(x => x > 10)); // 1

10. findLast(callback)
const arr = [5, 12, 8, 15];
console.log(arr.findLast(x => x > 10)); // 15

11. findLastIndex(callback)
const arr = [5, 12, 8, 15];
console.log(arr.findLastIndex(x => x > 10)); // 3

12. filter(callback)
const arr = [1, 2, 3, 4];
const pares = arr.filter(x => x % 2 === 0);
console.log(pares); // [2, 4]

13. some(callback)
const arr = [1, 2, 3];
console.log(arr.some(x => x > 2)); // true

14. every(callback)
const arr = [2, 4, 6];
console.log(arr.every(x => x % 2 === 0)); // true

15. includes(value)
const arr = ['a', 'b', 'c'];
console.log(arr.includes('b')); // true

16. indexOf(value)
const arr = [1, 2, 3, 2];
console.log(arr.indexOf(2)); // 1

17. lastIndexOf(value)
const arr = [1, 2, 3, 2];
console.log(arr.lastIndexOf(2)); // 3

18. reduce(callback)
const arr = [1, 2, 3, 4];
const suma = arr.reduce((acc, val) => acc + val, 0);
console.log(suma); // 10

19. reduceRight(callback)
const arr = ['a', 'b', 'c'];
const reversed = arr.reduceRight((acc, val) => acc + val, '');
console.log(reversed); // 'cba'

20. flat(depth)
const arr = [1, [2, [3]]];
console.log(arr.flat(2)); // [1, 2, 3]

21. flatMap(callback)
const arr = [1, 2, 3];
const duplicados = arr.flatMap(x => [x, x]);
console.log(duplicados); // [1, 1, 2, 2, 3, 3]

22. concat(...arrays)
const a = [1, 2];
const b = [3, 4];
console.log(a.concat(b)); // [1, 2, 3, 4]

23. push(...items)
const arr = [1, 2];
arr.push(3);
console.log(arr); // [1, 2, 3]

24. pop()
const arr = [1, 2, 3];
arr.pop();
console.log(arr); // [1, 2]

25. shift()
const arr = [1, 2, 3];
arr.shift();
console.log(arr); // [2, 3]

26. unshift(...items)
const arr = [2, 3];
arr.unshift(1);
console.log(arr); // [1, 2, 3]

27. splice(start, deleteCount, ...items)
const arr = [1, 2, 3];
arr.splice(1, 1, 99);
console.log(arr); // [1, 99, 3]

28. fill(value, start, end)
const arr = [1, 2, 3, 4];
arr.fill(0, 1, 3);
console.log(arr); // [1, 0, 0, 4]

29. copyWithin(target, start, end)
const arr = [1, 2, 3, 4];
arr.copyWithin(1, 2);
console.log(arr); // [1, 3, 4, 4]

30. reverse()
const arr = [1, 2, 3];
arr.reverse();
console.log(arr); // [3, 2, 1]

31. sort(compareFunction)
const arr = [3, 1, 2];
arr.sort((a, b) => a - b);
console.log(arr); // [1, 2, 3]

32. join(separator)
const arr = ['a', 'b', 'c'];
console.log(arr.join('-')); // 'a-b-c'

33. toString()
const arr = [1, 2, 3];
console.log(arr.toString()); // '1,2,3'

34. toLocaleString()
const arr = [123456.789];
console.log(arr.toLocaleString('de-DE')); // '123.456,789'
